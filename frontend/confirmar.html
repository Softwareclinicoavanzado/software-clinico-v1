// ============================================================ // confirmar-cita | ClinicOS // Procesa el clic del paciente en el correo (Confirmar/Cancelar). // Responde en JSON con un CÓDIGO (no texto ya traducido) — // confirmar.html es quien traduce ese código al idioma elegido. // ============================================================ const SUPABASE_URL = Deno.env.get("SUPABASE_URL"); const SUPABASE_SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); const CONFIRM_SECRET = Deno.env.get("CONFIRM_SECRET"); const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", }; async function generarToken(citaId: string, accion: string, secret: string): Promise { const data = new TextEncoder().encode(`${citaId}:${accion}:${secret}`); const hashBuffer = await crypto.subtle.digest("SHA-256", data); const hashArray = Array.from(new Uint8Array(hashBuffer)); return hashArray.map((b) => b.toString(16).padStart(2, "0")).join("").substring(0, 16); } function respuestaJSON(obj: Record, status: number): Response { return new Response(JSON.stringify(obj), { status, headers: { ...corsHeaders, "Content-Type": "application/json; charset=utf-8", "Cache-Control": "no-store, no-cache, must-revalidate, max-age=0", }, }); } async function registrarAuditoria(clinicaId: string, detalle: string) { try { await fetch(`${SUPABASE_URL}/rest/v1/auditoria`, { method: "POST", headers: { apikey: SUPABASE_SERVICE_ROLE_KEY!, Authorization: `Bearer ${SUPABASE_SERVICE_ROLE_KEY}`, "Content-Type": "application/json", Prefer: "return=minimal", }, body: JSON.stringify({ clinica_id: clinicaId, usuario_id: null, usuario_nombre: "Paciente (vía correo)", usuario_email: null, accion: "respuesta_paciente", entidad: "citas", detalle, }), }); } catch (_e) { /* no debe tumbar el flujo */ } } Deno.serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } try { if (!SUPABASE_URL || !SUPABASE_SERVICE_ROLE_KEY || !CONFIRM_SECRET) { return respuestaJSON({ exito: false, codigo: "config" }, 500); } const url = new URL(req.url); const id = url.searchParams.get("id"); const accion = url.searchParams.get("accion"); const token = url.searchParams.get("token"); if (!id || !accion || !token || !["confirmar", "cancelar"].includes(accion)) { return respuestaJSON({ exito: false, codigo: "invalido" }, 400); } const tokenEsperado = await generarToken(id, accion, CONFIRM_SECRET); if (token !== tokenEsperado) { return respuestaJSON({ exito: false, codigo: "invalido" }, 403); } const citaRes = await fetch( `${SUPABASE_URL}/rest/v1/citas?id=eq.${id}&select=id,estado,clinica_id,fecha,hora`, { headers: { apikey: SUPABASE_SERVICE_ROLE_KEY, Authorization: `Bearer ${SUPABASE_SERVICE_ROLE_KEY}` } } ); const citaData = await citaRes.json(); const cita = Array.isArray(citaData) && citaData.length > 0 ? citaData[0] : null; if (!cita) { return respuestaJSON({ exito: false, codigo: "no_encontrada" }, 404); } if (cita.estado === "cancelada" || cita.estado === "completada") { return respuestaJSON({ exito: false, codigo: "ya_inactiva" }, 200); } const nuevoEstado = accion === "confirmar" ? "confirmada" : "cancelada"; const updateRes = await fetch(`${SUPABASE_URL}/rest/v1/citas?id=eq.${id}`, { method: "PATCH", headers: { apikey: SUPABASE_SERVICE_ROLE_KEY, Authorization: `Bearer ${SUPABASE_SERVICE_ROLE_KEY}`, "Content-Type": "application/json", Prefer: "return=minimal", }, body: JSON.stringify({ estado: nuevoEstado }), }); if (!updateRes.ok) { return respuestaJSON({ exito: false, codigo: "error" }, 500); } await registrarAuditoria( cita.clinica_id, `Cita ${id} (${cita.fecha} ${cita.hora}) — paciente respondió: ${accion === "confirmar" ? "confirmó asistencia" : "canceló"}` ); return respuestaJSON( { exito: true, codigo: accion === "confirmar" ? "confirmada" : "cancelada" }, 200 ); } catch (err) { return respuestaJSON({ exito: false, codigo: "error" }, 500); } });
